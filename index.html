import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const SITE_URL = "https://kaminante7319-code.github.io/coro-liturgico-studio-web/"; const corsHeaders = { "Access-Control-Allow-Origin": "https://kaminante7319-code.github.io", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; Deno.serve(async (req: Request) => { const json = (body: Record, status = 200) => new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } if (req.method !== "POST") { return json({ error: "Método no permitido." }, 405); } try { const authHeader = req.headers.get("Authorization") || ""; if (!authHeader.startsWith("Bearer ")) { return json({ error: "Sesión no válida." }, 401); } const supabaseUrl = Deno.env.get("SUPABASE_URL"); const anonKey = Deno.env.get("SUPABASE_ANON_KEY"); const serviceRoleKey = Deno.env.get("SERVICE_ROLE_KEY"); if (!supabaseUrl || !anonKey || !serviceRoleKey) { console.error("Falta SUPABASE_URL, SUPABASE_ANON_KEY o SERVICE_ROLE_KEY."); return json({ error: "Falta configurar una variable secreta en Supabase.", }, 500); } const token = authHeader.slice("Bearer ".length); const authClient = createClient(supabaseUrl, anonKey, { global: { headers: { Authorization: `Bearer ${token}` } }, auth: { persistSession: false, autoRefreshToken: false, }, }); const { data: userData, error: userError } = await authClient.auth.getUser(); if (userError || !userData.user) { return json({ error: "No se pudo validar la sesión." }, 401); } const adminClient = createClient(supabaseUrl, serviceRoleKey, { auth: { persistSession: false, autoRefreshToken: false, }, }); const { data: adminRow, error: adminError } = await adminClient .from("admin_users") .select("user_id") .eq("user_id", userData.user.id) .maybeSingle(); if (adminError) { console.error("Error consultando admin_users:", adminError); return json({ error: "No se pudo verificar el permiso de administrador.", detail: adminError.message, code: adminError.code || null, }, 500); } if (!adminRow) { return json({ error: "La sesión es válida, pero este usuario no está registrado como administrador.", }, 403); } const body = await req.json().catch(() => ({})); const email = String(body.email || "").trim().toLowerCase(); if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) { return json({ error: "Escribe un correo electrónico válido." }, 400); } // Buscar si el correo ya tiene una cuenta en Supabase Auth. // No se elimina ni se duplica una cuenta existente. let existingUser: { id: string; email?: string } | null = null; const perPage = 1000; for (let page = 1; page <= 10; page++) { const { data, error } = await adminClient.auth.admin.listUsers({ page, perPage, }); if (error) { console.error("Error buscando usuario en Auth:", error); return json({ error: "No se pudo comprobar si el correo ya tiene una cuenta.", detail: error.message, }, 500); } existingUser = data.users.find( (user) => (user.email || "").toLowerCase() === email, ) || null; if (existingUser || data.users.length < perPage) break; } if (existingUser) { // Reactivar la autorización conservando el mismo user_id. const { error: upsertError } = await adminClient .from("authorized_users") .upsert( { user_id: existingUser.id, email, active: true, }, { onConflict: "user_id" }, ); if (upsertError) { console.error("Error reactivando authorized_users:", upsertError); return json({ error: "La cuenta existe, pero no se pudo reactivar su autorización.", detail: upsertError.message, }, 500); } // Para una cuenta existente se envía un enlace de acceso (magic link). // inviteUserByEmail no sirve para volver a invitar una cuenta ya creada. const publicClient = createClient(supabaseUrl, anonKey, { auth: { persistSession: false, autoRefreshToken: false, }, }); const { error: accessError } = await publicClient.auth.signInWithOtp({ email, options: { shouldCreateUser: false, emailRedirectTo: SITE_URL, }, }); if (accessError) { console.error("Error enviando enlace de acceso:", accessError); return json({ ok: false, reactivated: true, email, user_id: existingUser.id, error: "La autorización quedó reactivada, pero no se pudo enviar el enlace de acceso.", detail: accessError.message, }, 502); } return json({ ok: true, reactivated: true, invitation_sent: false, access_link_sent: true, email, user_id: existingUser.id, message: "La cuenta existente fue reactivada y se envió un nuevo enlace de acceso al mismo correo.", }); } // Si no existe, crear una invitación nueva. const { data: inviteData, error: inviteError } = await adminClient.auth.admin.inviteUserByEmail(email, { redirectTo: SITE_URL, }); if (inviteError) { console.error("Error enviando invitación:", inviteError); return json({ error: inviteError.message }, 400); } if (!inviteData.user) { return json({ error: "Supabase no devolvió la cuenta invitada.", }, 500); } const { error: upsertError } = await adminClient .from("authorized_users") .upsert( { user_id: inviteData.user.id, email, active: true, }, { onConflict: "user_id" }, ); if (upsertError) { console.error("Error registrando autorización:", upsertError); return json({ error: "La invitación fue enviada, pero no se pudo registrar la autorización.", detail: upsertError.message, invitation_sent: true, user_id: inviteData.user.id, }, 500); } return json({ ok: true, reactivated: false, invitation_sent: true, email, user_id: inviteData.user.id, message: "Invitación enviada y autorización registrada correctamente.", }); } catch (error) { console.error("Error inesperado en invite-members:", error); return json({ error: error instanceof Error ? error.message : "Error interno.", }, 500); } });
